filter()

示例
先过滤再渲染
const items = users.filter((item) => item.gender == "male")
  .map((item) => <li key={item.id}>{item.name}</li>);

Rendering Object

无法直接渲染对象
将其转换为键 key、值 value 或键值后再渲染
示例
const address={ district: "xxx", city: "ay", province: "gx", code: "455112" }
使用值 value；key 留空
<div>{Object.entries(address).map(([, val]) => val)}</div>

import "./demo.css";
interface Address {
  district: string;
  city: string;
  province: string;
  code: string;
}

interface User {
  id: number;
  name: string;
  age: number;
  gender: string;
  address: Address;
  msg: number[];
}

function Demo() {
  const users: User[] = [
    { id: 0, name: "gl", age: 18, gender: "male", address: { district: "xxx", city: "ay", province: "gx", code: "455112" }, msg: [0, 1, 2] },
    { id: 1, name: "gz", age: 15, gender: "female", address: { district: "yyy", city: "gl", province: "gd", code: "541000" }, msg: [0, 1, 2] },
    { id: 2, name: "sh", age: 28, gender: "male", address: { district: "zzz", city: "sh", province: "js", code: "200240" }, msg: [0, 1, 2] },
  ];

  const getMsg = (list: number[]) => {
    return list.length ? list.map((item) => <li key={item}>{item}</li>) : "暂无";
  };

  const getAddress = (obj: Address) => {
    return Object.entries(obj).map(([key, val]) => <li key={key}>{val}</li>);
  };

  const getGender = (gender: string) => (gender == "male" ? "男" : "女");

  const getAge = (age: number) => (age > 18 ? "成年" : "未成年");

  const items = users.map((item) => {
    return (
      <li key={item.id}>
        <p>Name：{item.name}</p>
        <p>Age：{getAge(item.age)}</p>
        <p>Gender：{getGender(item.gender)}</p>
        <p>Address：</p>
        <ul>{getAddress(item.address)}</ul>
        <p>Message：</p>
        <ul>{item.msg.length ? getMsg(item.msg) : "暂无"}</ul>
      </li>
    );
  });

  return (
    <>
      <ul>{items}</ul>
    <>
  );
}
export default Demo;
