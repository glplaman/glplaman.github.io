处理 Excel 的一般流程
  1. 读取
  2. 编辑
  3. 更新
  4. 输出
  5. 关闭
# 导入包
import pandas as pd

# 读取excel文件
res=pd.read_excel('./data/sample.xlsx')

# 处理某个字段并定义新字段接受结果
res['msg']=res['test'].str.split('/').str[0].str.strip()

# 创建写文件
file=pd.ExcelWriter('./data/test.xlsx')

# 将结果写入文件指定的工作表
res['msg'].to_excel(file, sheet_name='msg')

# 关闭文件
file.close()
