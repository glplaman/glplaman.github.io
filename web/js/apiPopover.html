[] 按下 h/H 查看或关闭帮助
获取元素实例并分配方法
const popover = document.querySelector("#popover-filter");
document.addEventListener("keydown", (e) => {
  if (e.key === "h" || e.key === "H") {
    popover.togglePopover();
  }
});

const popoverEvent = document.querySelector("#popover-event");
popoverEvent.addEventListener("toggle", (e) => {
  if (popoverEvent.matches(":popover-open")) {
    setTimeout(() => {
      popoverEvent.hidePopover();
    }, 3000);
  }
});
